\section*{\color{ecogreen} II. TỐI ƯU HÓA TOÁN HỌC (OPTIMIZATION)}
\textit{Giả sử muốn giữ nguyên $V$, tìm kích thước để diện tích nhỏ nhất.}

\begin{itemize}
    \item \textbf{Công thức tối ưu:} $h = 2r$.
    \item \textbf{Bán kính tối ưu ($r_{min}$):}
    $$ 2\pi r^3 = V \Rightarrow r_{min} = \sqrt[3]{\frac{V}{2\pi}} = \sqrt[3]{\frac{\dots\dots}{2\pi}} \approx \dots\dots\dots (cm) $$
    \item \textbf{Chiều cao tối ưu ($h_{min}$):}
    $$ h_{min} = 2 \times r_{min} \approx \dots\dots\dots (cm) $$
    \item \textbf{Diện tích tối ưu ($S_{min}$):}
    $$ S_{min} = 2\pi r_{min}^2 + 2\pi r_{min}h_{min} \approx \dots\dots\dots (cm^2) $$
\end{itemize}
