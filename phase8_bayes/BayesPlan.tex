\begin{sectionbox}{III. TIẾN TRÌNH DẠY HỌC (MÔ HÌNH 5E)}

\phantomsection\subsection*{1. ENGAGE (Gắn kết - 10p): Cú sốc nhận thức}\addcontentsline{toc}{subsection}{1. ENGAGE (Gắn kết - 10p)}
\textbf{Hoạt động:} "Bản án của AI"
\begin{itemize}
    \item \textbf{GV nêu tình huống:} "Một hệ thống AI chẩn đoán ung thư mới được công bố với độ chính xác lên tới \textbf{99\%}. Bệnh X cực hiếm, chỉ 0.1\% dân số mắc phải. Hôm nay, AI báo anh Nam \textbf{DƯƠNG TÍNH (+)}."
    \item \textbf{Câu hỏi thảo luận:} Theo em, xác suất anh Nam thực sự bị bệnh là bao nhiêu?
    \item \textbf{Dự đoán:} GV mở Poll bình chọn (A: >90\%, B: 50-90\%, C: <10\%). Đa số HS sẽ chọn A do tâm lý "tin vào con số 99\%".
    \item \textbf{Dẫn dắt:} "Liệu trực giác của chúng ta có đúng? Hãy cùng kiểm chứng bằng Dữ liệu lớn (Big Data)."
\end{itemize}

\phantomsection\subsection*{2. EXPLORE (Khám phá - 20p): Truy tìm sự thật}\addcontentsline{toc}{subsection}{2. EXPLORE (Khám phá - 20p)}
\textbf{Hoạt động:} Thực nghiệm trên \textbf{BayesHunter App}
\begin{enumerate}
    \item HS truy cập: \href{https://stem-1h8.pages.dev/bayes.html}{\texttt{stem-1h8.pages.dev/bayes.html}}
    \item \textbf{Thiết lập tham số:}
    \begin{itemize}
        \item Tỷ lệ bệnh (Prevalence): $0.1\%$
        \item Độ nhạy (Sensitivity): $99\%$
        \item Độ đặc hiệu (Specificity): $99\%$
    \end{itemize}
    \item \textbf{Quan sát Dot Matrix (1000 người):}
    \begin{itemize}
        \item Tìm số chấm \textbf{Đỏ} (Người bệnh thật): Rất ít ($\approx 1$ người).
        \item Tìm số chấm có \textbf{Vòng tròn} (AI báo Dương tính): Khoảng 10-11 người.
        \item Đếm số người \textbf{Vừa Đỏ Vừa có Vòng} (Dương tính thật).
    \end{itemize}
    \item \textbf{Ghi chép:} HS điền số liệu vào Phiếu học tập. Nhận ra rằng trong 11 người bị AI báo bệnh, chỉ có 1 người bệnh thật!
\end{enumerate}

\phantomsection\subsection*{3. EXPLAIN (Giải thích - 30p): Sức mạnh của Toán học}\addcontentsline{toc}{subsection}{3. EXPLAIN (Giải thích - 30p)}
\textbf{Hoạt động:} Giải mã bằng Sơ đồ cây (Tree Diagram)
\begin{center}
\begin{tikzpicture}[level distance=2.5cm,
  level 1/.style={sibling distance=6cm},
  level 2/.style={sibling distance=3cm}]
  \node {Dân số (10.000 người)}
    child {node {Bệnh (10 người)}
      child {node {\textbf{Dương tính thật} (9.9 $\approx$ 10)} edge from parent node[left] {99\%}}
      child {node {Âm tính giả (0.1 $\approx$ 0)} edge from parent node[right] {1\%}}
      edge from parent node[left] {0.1\%}
    }
    child {node {Khỏe (9.990 người)}
      child {node {\textbf{Dương tính giả} (99.9 $\approx$ 100)} edge from parent node[left] {1\%}}
      child {node {Âm tính thật (9.890)} edge from parent node[right] {99\%}}
      edge from parent node[right] {99.9\%}
    };
\end{tikzpicture}
\end{center}

\begin{mathbox}{Công thức Bayes}
$$ P(\text{Bệnh}|+) = \frac{P(\text{Bệnh} \cap +)}{P(+)} = \frac{\text{Số ca Dương tính thật}}{\text{Tổng số ca Dương tính}} $$
Thay số từ sơ đồ cây:
$$ P(\text{Bệnh}|+) = \frac{10}{10 + 100} = \frac{10}{110} \approx \mathbf{9\%} $$
\textbf{Kết luận:} Dù AI chính xác 99\%, nhưng vì bệnh quá hiếm, số lượng người khỏe bị báo nhầm (Dương tính giả) áp đảo số người bệnh thật.
\end{mathbox}

\phantomsection\subsection*{4. ELABORATE (Áp dụng - 20p): Kỹ sư AI tài ba}\addcontentsline{toc}{subsection}{4. ELABORATE (Áp dụng - 20p)}
\textbf{Hoạt động:} Tối ưu hóa hệ thống
GV đặt vấn đề: "Làm sao để tăng độ tin cậy từ 9\% lên trên 50\%?"
HS làm việc nhóm, thử nghiệm trên App:
\begin{itemize}
    \item \textbf{Cách 1:} Tăng Độ đặc hiệu lên 99.9\% (AI siêu việt). $\to$ Khó khả thi về kỹ thuật.
    \item \textbf{Cách 2:} Tăng tỷ lệ bệnh ($P(B)$) bằng cách \textbf{Sàng lọc đối tượng}. Chỉ xét nghiệm những người có triệu chứng hoặc tiền sử gia đình (nhóm nguy cơ cao, ví dụ tỷ lệ bệnh là 10\%).
    \item \textbf{Thử nghiệm lại:} Với $P(B)=10\%$, kết quả PPV tăng vọt lên $>90\%$.
\end{itemize}

\phantomsection\subsection*{5. EVALUATE (Đánh giá - 10p)}\addcontentsline{toc}{subsection}{5. EVALUATE (Đánh giá - 10p)}
HS hoàn thành báo cáo:
1. Giải thích tại sao không nên xét nghiệm đại trà (mass screening) cho các bệnh hiếm.
2. Viết một đoạn văn ngắn tư vấn cho bệnh nhân vừa nhận kết quả Dương tính để họ bình tĩnh.
\end{sectionbox}

\phantomsection\subsection*{Phụ lục 1: Thuật ngữ Y tế}\addcontentsline{toc}{subsection}{Phụ lục 1: Thuật ngữ Y tế}
\begin{itemize}
    \item \textbf{Sensitivity (Độ nhạy):} Bệnh thật $\to$ Test ra (+).

    \item \textbf{PPV (Positive Predictive Value):} Test ra (+) $\to$ Bệnh thật.
\end{itemize}
