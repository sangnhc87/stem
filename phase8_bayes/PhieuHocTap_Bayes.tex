\section*{\color{medblue} PHẦN 1: THẨM ĐỊNH MỘT "SIÊU AI" (99\% CHÍNH XÁC)}
\textbf{Dữ liệu đầu vào:}
- Tỷ lệ mắc bệnh hiểm nghèo X trong dân số: $0.1\%$ ($P(B) = 0.001$).
- Độ chính xác của AI (Nhạy \& Đặc hiệu): $99\%$ ($0.99$).

\textbf{Mô phỏng 10,000 người dân:}
\begin{enumerate}
    \item \textbf{Số người Bệnh thực tế:} $10,000 \times 0.001 = \dots\dots\dots$ người.
    \item \textbf{Số người Khỏe thực tế:} $10,000 - \dots\dots = \dots\dots\dots$ người.
    \item \textbf{Với người Bệnh, AI phát hiện đúng (Dương tính thật):}
    $$ \dots\dots\dots \times 0.99 \approx \dots\dots\dots \text{ người.} $$
    \item \textbf{Với người Khỏe, AI báo nhầm (Dương tính giả):}
    $$ \dots\dots\dots \times (100\% - 99\%) = \dots\dots\dots \times 0.01 \approx \dots\dots\dots \text{ người.} $$
    \item \textbf{Tổng số ca AI báo Dương tính:}
    $$ \dots\dots\dots + \dots\dots\dots = \dots\dots\dots \text{ ca.} $$
\end{enumerate}
